# The held light with ray-traced block light: a dark hall at night, a torch in hand next to a
# pillar and a fence; lightmap (A), ray traced (B), and after a few steps (C)
sleep 3
world lookrthand 42 flat
waitWorld
fly
cmd /gamerule doDaylightCycle false
cmd /gamerule doMobSpawning false
cmd /time set 18000
config heldLight true
rtlighting 2
config rtBlockLight false
cmd /tp @p -985 4 -985 180 20
sleep 4
cmd /kill @e[type=!Player]
cmd /fill -1001 2 -1001 -969 12 -969 stonebrick
cmd /fill -1000 4 -1000 -970 10 -970 air
cmd /fill -1000 3 -1000 -970 3 -970 quartz_block
cmd /fill -986 4 -988 -986 7 -988 stonebrick
cmd /fill -983 4 -989 -981 4 -989 fence
cmd /clear @p
cmd /give @p torch 1
slot 0
cmd /tp @p -985 4 -985 180 20
sleep 8
capture CAPTURE_A
config rtBlockLight true
sleep 4
capture CAPTURE_B
cmd /tp @p -984 4 -986 160 25
sleep 2
capture CAPTURE_C
quit
