# Ray-traced sky light in the forest: lightmap (A) and traced (B) with GI ray traced; then
# with GI off (sky-only rays): lightmap (C) and traced (D)
sleep 3
fancy true
world ref1 1234567
waitWorld
cmd /gamerule doDaylightCycle false
cmd /gamerule doMobSpawning false
cmd /weather clear 100000
cmd /time set 6000
cmd /kill @e[type=!Player]
cmd /tp @p 64 78 -32 135 10
rtlighting 1
sleep 20
capture CAPTURE_A
config rtSkyLight true
sleep 5
capture CAPTURE_B
config rtSkyLight false
config globalIllumination 0
sleep 5
capture CAPTURE_C
config rtSkyLight true
sleep 5
capture CAPTURE_D
quit
